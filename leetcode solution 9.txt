class Solution {
    public int lengthOfLastWord(String s) {
        int i = s.length() - 1; // Начинаем с последнего символа строки

        while (s.charAt(i) == ' ') { // Пока текущий символ является пробелом
            i--; // Двигаемся влево, пропуская пробелы
        }

        int length = 0; // Создаём переменную для подсчёта длины последнего слова

        while (i >= 0 && s.charAt(i) != ' ') { // Идём влево, пока не встретим пробел
            length++; // Увеличиваем длину слова на 1
            i--; // Переходим к предыдущему символу
        }

        return length; // Возвращаем длину последнего слова
    }
}
